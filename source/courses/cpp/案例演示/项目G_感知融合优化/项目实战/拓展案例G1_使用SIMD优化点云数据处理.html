情境背景

激光雷达是许多高级智能系统的核心传感器，它通过发射激光束感知周围环境，生成由数百万个三维点组成的“点云”数据。对这些海量点云数据进行实时处理（如坐标变换、滤波、聚类）是实现环境感知和建图定位功能的关键。然而，逐点进行数学运算的传统编程方式，在面对每秒数百万甚至上千万个点时，会成为整个系统的性能瓶颈。现代 CPU 提供了一种强大的并行计算技术——单指令多数据流（SIMD），允许 CPU 在一个时钟周期内使用一条指令同时对多个数据执行相同的操作，可将计算密集型的循环操作性能提升数倍。

案例要求

对大量三维点云执行整体平移，分别实现标准 C++ 循环和跨平台 SIMD 版本并比较效率。定义含 float x, y, z 与 padding 的 16 字节对齐 Point；translate_points_scalar 用普通 for 循环逐点平移，translate_points_simd 按处理器架构使用 ARM NEON 或 x86 SSE 每次处理 4 个点，并为其他架构提供标量回退。生成大规模点云，分别计时并验证两种实现的坐标结果一致，不预设固定加速倍数。
源码核对重点：源码按平台选 ARM NEON、x86 SSE 或标量回退，并处理不足一组的尾部；加速效果必须以实测为准。

#include <cmath>
#include <chrono>
#include <iostream>
#include <vector>

#if defined(__aarch64__) || defined(__ARM_NEON) || defined(__ARM_NEON__)
#include <arm_neon.h>
#define USE_NEON
#elif defined(__SSE__)
#include <immintrin.h>
#define USE_SSE
#endif

// 16字节结构与128位寄存器宽度一致
struct alignas(16) Point {
    float x, y, z, padding;
};

void translate_points_scalar(std::vector<Point>& points, float tx, float ty, float tz) {
    for (Point& point : points) {
        point.x += tx;
        point.y += ty;
        point.z += tz;
    }
}

void translate_points_simd(std::vector<Point>& points, float tx, float ty, float tz) {
#if defined(USE_NEON)
    const float32x4_t offset_x = vdupq_n_f32(tx);
    const float32x4_t offset_y = vdupq_n_f32(ty);
    const float32x4_t offset_z = vdupq_n_f32(tz);
    std::size_t index = 0;
    for (; index + 3 < points.size(); index += 4) {
        // 交错加载得到4个x、4个y、4个z和4个padding
        float32x4x4_t values = vld4q_f32(&points[index].x);
        values.val[0] = vaddq_f32(values.val[0], offset_x);
        values.val[1] = vaddq_f32(values.val[1], offset_y);
        values.val[2] = vaddq_f32(values.val[2], offset_z);
        vst4q_f32(&points[index].x, values);
    }
#elif defined(USE_SSE)
    const __m128 offset_x = _mm_set1_ps(tx);
    const __m128 offset_y = _mm_set1_ps(ty);
    const __m128 offset_z = _mm_set1_ps(tz);
    std::size_t index = 0;
    for (; index + 3 < points.size(); index += 4) {
        __m128 point0 = _mm_load_ps(&points[index].x);
        __m128 point1 = _mm_load_ps(&points[index + 1].x);
        __m128 point2 = _mm_load_ps(&points[index + 2].x);
        __m128 point3 = _mm_load_ps(&points[index + 3].x);
        _MM_TRANSPOSE4_PS(point0, point1, point2, point3);
        point0 = _mm_add_ps(point0, offset_x);
        point1 = _mm_add_ps(point1, offset_y);
        point2 = _mm_add_ps(point2, offset_z);
        _MM_TRANSPOSE4_PS(point0, point1, point2, point3);
        _mm_store_ps(&points[index].x, point0);
        _mm_store_ps(&points[index + 1].x, point1);
        _mm_store_ps(&points[index + 2].x, point2);
        _mm_store_ps(&points[index + 3].x, point3);
    }
#else
    std::size_t index = 0;
#endif
    // 处理不足4个的尾部元素，或在不支持内建函数的平台执行标量回退
    for (; index < points.size(); ++index) {
        points[index].x += tx;
        points[index].y += ty;
        points[index].z += tz;
    }
}

int main() {
    const std::size_t point_count = 500000;
    std::vector<Point> scalar_points(point_count);

    for (std::size_t index = 0; index < point_count; ++index) {
        const float value = static_cast<float>(index) * 0.01f;
        scalar_points[index] = {value, value * 2.0f, value * 3.0f, 0.0f};
    }
    std::vector<Point> simd_points = scalar_points;

    auto start_scalar = std::chrono::high_resolution_clock::now();
    translate_points_scalar(scalar_points, 1.5f, -2.5f, 3.0f);
    auto end_scalar = std::chrono::high_resolution_clock::now();

    auto start_simd = std::chrono::high_resolution_clock::now();
    translate_points_simd(simd_points, 1.5f, -2.5f, 3.0f);
    auto end_simd = std::chrono::high_resolution_clock::now();

    bool results_match = true;
    for (std::size_t index = 0; index < point_count; ++index) {
        if (std::fabs(scalar_points[index].x - simd_points[index].x) > 0.0001f ||
            std::fabs(scalar_points[index].y - simd_points[index].y) > 0.0001f ||
            std::fabs(scalar_points[index].z - simd_points[index].z) > 0.0001f) {
            results_match = false;
            break;
        }
    }

    const double scalar_ms = std::chrono::duration<double, std::milli>(
        end_scalar - start_scalar).count();
    const double simd_ms = std::chrono::duration<double, std::milli>(
        end_simd - start_simd).count();
    std::cout << "标量耗时: " << scalar_ms << " ms\n"
              << "SIMD耗时: " << simd_ms << " ms\n"
              << "结果校验: " << (results_match ? "一致" : "不一致") << std::endl;

    return results_match ? 0 : 1;
}
