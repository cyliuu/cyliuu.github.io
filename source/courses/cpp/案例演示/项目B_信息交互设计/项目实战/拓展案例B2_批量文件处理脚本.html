情境背景

eee_B2/addition_validator.cpp
#include <iostream>

int main() {
    int num1, num2, expected_sum;
    int test_count = 0, pass_count = 0;

    std::cout << "--- 开始自动化验证测试 ---" << std::endl;
    while (std::cin >> num1 >> num2 >> expected_sum) {
        test_count++;
        int actual_sum = num1 + num2;
        std::cout << "[用例 " << test_count << "]: " << num1 << " + " << num2 << std::endl;
        if (actual_sum == expected_sum) {
            std::cout << "  - 结果: [PASS]" << std::endl;
            pass_count++;
        } else {
            std::cout << "  - 结果: [FAIL]" << std::endl;
        }
    }
    std::cout << "总用例数: " << test_count << " 通过: " << pass_count
              << " 失败: " << test_count - pass_count << std::endl;
    return 0;
}
