情境背景

摄像头是许多智能系统的“眼睛”，尤其是在自动驾驶领域。为了让程序能够处理视觉信息，第一步就是必须能将摄像头捕捉到的图像加载到内存中并显示出来，以便进行调试和分析。直接从零开始编写代码来解码 JPEG、PNG 等复杂的图像格式是一项极其艰巨的任务。幸运的是，我们不必“重复造轮子”。OpenCV（Open Source Computer Vision Library） 是全球最著名的计算机视觉库，它为我们提供了一套强大而简洁的工具，可以轻松完成图像的读取、处理和显示。本案例将是你进入计算机视觉世界的第一步，体验如何利用这个工业级的第三方库轻松地为应用程序增添“视觉”能力。

案例要求

配置并使用 OpenCV 库，编写程序从本地磁盘加载道路图片，并在图形窗口中显示。案例目录已提供 road.jpg，CMake 构建时通过 ROAD_IMAGE_PATH 注入绝对路径。核心要求是：使用 cv::imread 加载图片并检查结果，使用 cv::imshow 显示图像，使用 cv::waitKey 等待用户按键；若按下 S，则将当前图像另存为 saved_road.png。
源码核对重点：源码先检查读取结果；网页解释 imread/imshow/waitKey/S 保存流程，不在浏览器声称运行 OpenCV GUI。

// 使用OpenCV加载并显示摄像头图像

// 包含 OpenCV 的主头文件
//    这一个头文件通常就包含了大部分常用功能
#include <opencv2/opencv.hpp>
#include <iostream>

#ifndef ROAD_IMAGE_PATH
#define ROAD_IMAGE_PATH "road.jpg"
#endif

int main() {
    // CMake 构建时会注入图片的绝对路径；未注入时使用相对路径
    std::string image_path = ROAD_IMAGE_PATH;

    // 使用 cv::imread 函数读取图像
    //    它会自动解码图片文件，并将像素数据存入一个 cv::Mat 对象中
    cv::Mat image = cv::imread(image_path);

    // 检查图像是否加载成功
    //    如果路径错误或文件损坏，cv::imread会返回一个空的Mat对象
    if (image.empty()) {
        std::cerr << "错误: 无法加载图像 '" << image_path << "'" << std::endl;
        std::cerr << "请检查图片路径、文件格式和读取权限。" << std::endl;
        return -1; // 异常退出
    }

    // 使用 cv::imshow 在一个窗口中显示图像
    //    第一个参数 "Camera View" 是窗口的标题
    //    第二个参数是要显示的 cv::Mat 对象
    cv::imshow("Camera View", image);

    // 等待用户按键，使窗口保持显示
    //    cv::waitKey(0) 会让程序在此处暂停，无限期地等待键盘输入
    //    cv::waitKey 使窗口保持显示；如果没有这一行，
    //    程序会一闪而过，你根本看不到窗口
    int key = cv::waitKey(0);

    // (可选) 根据按键执行操作
    if (key == 's' || key == 'S') {
        cv::imwrite("saved_road.png", image);
        std::cout << "图像已另存为 saved_road.png" << std::endl;
    }

    return 0;
}
eee_D2/CMakeLists.txt
cmake_minimum_required(VERSION 3.16)
project(eee_D2 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(OpenCV QUIET COMPONENTS core imgcodecs highgui)

if(NOT OpenCV_FOUND)
    message(STATUS "OpenCV unavailable: skipping optional eee_D2 target")
    return()
endif()

add_executable(eee_D2 main.cpp)
target_link_libraries(eee_D2 PRIVATE ${OpenCV_LIBS})
target_include_directories(eee_D2 PRIVATE ${OpenCV_INCLUDE_DIRS})
target_compile_definitions(eee_D2 PRIVATE
    ROAD_IMAGE_PATH="${CMAKE_CURRENT_SOURCE_DIR}/road.jpg")
